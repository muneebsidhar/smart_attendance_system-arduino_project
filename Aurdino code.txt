#include <SPI.h>
#include <MFRC522.h>
#include <Wire.h>
#include <LiquidCrystal_I2C.h>

#define SS_PIN 10
#define RST_PIN 9

MFRC522 rfid(SS_PIN, RST_PIN);
LiquidCrystal_I2C lcd(0x27,16,2);

const int greenLED = 7;
const int buzzer = 6;

// Student Database
struct Student {
  String uid;
  String roll;
  String name;
};

Student students[] = {
  {"7DB72C05", "01", "Zara Zafar"},
  {"5FF1CDC8", "02", "Laiba Noor"},
  {"11223344", "103", "Bilal Raza"}
};
const int studentCount = sizeof(students)/sizeof(students[0]);

// Clear LCD quickly
void fastClearLCD() {
  lcd.setCursor(0,0); lcd.print("                ");
  lcd.setCursor(0,1); lcd.print("                ");
}

void setup() {
  Serial.begin(9600);
  SPI.begin();
  rfid.PCD_Init();

  lcd.init();
  lcd.backlight();

  pinMode(greenLED, OUTPUT);
  pinMode(buzzer, OUTPUT);

  lcd.setCursor(0,0);
  lcd.print("Place your card");
  lcd.setCursor(0,1);
  lcd.print("on the reader");
}

void loop() {
  if(!rfid.PICC_IsNewCardPresent()) return;
  if(!rfid.PICC_ReadCardSerial()) return;

  // Read UID
  String uid = "";
  for(byte i=0;i<rfid.uid.size;i++){
    char buffer[3];
    sprintf(buffer,"%02X",rfid.uid.uidByte[i]);
    uid += buffer;
  }
  uid.toUpperCase();

  bool found = false;

  for(int i=0;i<studentCount;i++){
    if(uid == students[i].uid){
      // LED Blink
      digitalWrite(greenLED,HIGH);
      delay(200);
      digitalWrite(greenLED,LOW);

      // Show on LCD
      fastClearLCD();
      lcd.setCursor(0,0);
      lcd.print("Roll: " + students[i].roll);
      lcd.setCursor(0,1);
      lcd.print(students[i].name);
      delay(1000); // show 1 second

      // Serial output for Excel (Roll, Name, UID, Status)
      Serial.print(students[i].roll); Serial.print(",");
      Serial.print(students[i].name); Serial.print(",");
      Serial.print(uid); Serial.print(",");
      Serial.println("Present");

      found = true;
      break;
    }
  }

  if(!found){
    // Unknown card alert
    for(int i=0;i<2;i++){
      tone(buzzer,1000);
      delay(150);
      noTone(buzzer);
      delay(150);
    }

    fastClearLCD();
    lcd.setCursor(0,0); lcd.print("Unknown Card");
    lcd.setCursor(0,1); lcd.print("Access Denied");
    delay(1000);

    // Serial output for Excel
    Serial.print("N/A,Unknown,"); 
    Serial.print(uid); Serial.print(",");
    Serial.println("Unknown Card");
  }

  // Reset LCD
  fastClearLCD();
  lcd.setCursor(0,0); lcd.print("Place your card");
  lcd.setCursor(0,1); lcd.print("on the reader");

  rfid.PICC_HaltA();
  rfid.PCD_StopCrypto1();
  delay(50);
}
